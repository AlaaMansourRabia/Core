## Usage

All charts use the `ChartContainer` wrapper and the `useChartTheme` hook for theme-aware colors:

```tsx
import {ChartContainer, useChartTheme} from "@wakecap/core-ui/chart";

function MyChart() {
	const {colors, textColor, borderColor} = useChartTheme();

	const option = {
		xAxis: {type: "category", data: ["Mon", "Tue", "Wed"]},
		yAxis: {type: "value"},
		series: [{type: "bar", data: [120, 200, 150], itemStyle: {color: colors[0]}}],
	};

	return <ChartContainer option={option} style={{height: 300}} />;
}
```
